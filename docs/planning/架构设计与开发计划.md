### 2. 数据库表结构设计

#### 2.1 频道表 (channels)
```sql
CREATE TABLE channels (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name VARCHAR(100) NOT NULL,
    channel_id VARCHAR(100) NOT NULL UNIQUE,
    url VARCHAR(255) NOT NULL,
    cpm DECIMAL(5,2) NOT NULL DEFAULT 1.00,
    category VARCHAR(50),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    is_active BOOLEAN DEFAULT TRUE
);
```

#### 2.2 频道数据表 (channel_data)
```sql
CREATE TABLE channel_data (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    channel_id INTEGER NOT NULL,
    date DATE NOT NULL,
    subscribers INTEGER NOT NULL,
    total_views INTEGER NOT NULL,
    video_count INTEGER NOT NULL,
    subscriber_change INTEGER DEFAULT 0,
    view_change INTEGER DEFAULT 0,
    video_change INTEGER DEFAULT 0,
    estimated_revenue DECIMAL(10,2) DEFAULT 0.00,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (channel_id) REFERENCES channels(id) ON DELETE CASCADE,
    UNIQUE(channel_id, date)
);
```

#### 2.3 排行榜视图设计
```sql
-- 日收入排行榜
CREATE VIEW daily_revenue_ranking AS
SELECT 
    c.name,
    c.channel_id,
    cd.estimated_revenue,
    ROW_NUMBER() OVER (ORDER BY cd.estimated_revenue DESC) as rank
FROM channels c
JOIN channel_data cd ON c.id = cd.channel_id
WHERE cd.date = date('now') AND c.is_active = TRUE
ORDER BY cd.estimated_revenue DESC;

-- 月收入排行榜
CREATE VIEW monthly_revenue_ranking AS
SELECT 
    c.name,
    c.channel_id,
    SUM(cd.estimated_revenue) as total_revenue,
    ROW_NUMBER() OVER (ORDER BY SUM(cd.estimated_revenue) DESC) as rank
FROM channels c
JOIN channel_data cd ON c.id = cd.channel_id
WHERE cd.date >= date('now', '-30 days') AND c.is_active = TRUE
GROUP BY c.id, c.name, c.channel_id
ORDER BY total_revenue DESC;

-- 日订阅增长排行榜
CREATE VIEW daily_subscriber_ranking AS
SELECT 
    c.name,
    c.channel_id,
    cd.subscriber_change,
    ROW_NUMBER() OVER (ORDER BY cd.subscriber_change DESC) as rank
FROM channels c
JOIN channel_data cd ON c.id = cd.channel_id
WHERE cd.date = date('now') AND c.is_active = TRUE
ORDER BY cd.subscriber_change DESC;

-- 月订阅增长排行榜
CREATE VIEW monthly_subscriber_ranking AS
SELECT 
    c.name,
    c.channel_id,
    SUM(cd.subscriber_change) as total_subscriber_change,
    ROW_NUMBER() OVER (ORDER BY SUM(cd.subscriber_change) DESC) as rank
FROM channels c
JOIN channel_data cd ON c.id = cd.channel_id
WHERE cd.date >= date('now', '-30 days') AND c.is_active = TRUE
GROUP BY c.id, c.name, c.channel_id
ORDER BY total_subscriber_change DESC;
```

```python
class DataScheduler:
    def __init__(self):
        self.scheduler = BlockingScheduler()
    
    def setup_daily_scraping(self):
        """设置每日数据抓取"""
        self.scheduler.add_job(
            func=self.scrape_all_channels,
            trigger='cron',
            hour=8,
            minute=0,
            id='daily_scraping'
        )
    
    def scrape_all_channels(self):
        """抓取所有频道数据"""
        channels = Channel.query.filter_by(is_active=True).all()
        for channel in channels:
            try:
                self.scrape_channel_data(channel)
            except Exception as e:
                logger.error(f"抓取频道 {channel.name} 数据失败: {e}")
```

```python
def calculate_incremental_data(channel_id, new_data):
    """
    计算增量数据的核心算法
    """
    # 获取昨天的数据
    yesterday = ChannelData.query.filter_by(
        channel_id=channel_id,
        date=date.today() - timedelta(days=1)
    ).first()
    
    if not yesterday:
        # 如果没有昨天数据，设置变化量为0
        return {
            'subscriber_change': 0,
            'view_change': 0,
            'video_change': 0,
            'estimated_revenue': 0
        }
    
    # 计算各项变化
    subscriber_change = new_data['subscribers'] - yesterday.subscribers
    view_change = new_data['total_views'] - yesterday.total_views
    video_change = new_data['video_count'] - yesterday.video_count
    
    # 计算预估收入：浏览量增量 * CPM / 1000
    channel = Channel.query.get(channel_id)
    estimated_revenue = max(0, view_change * channel.cpm / 1000)
    
    return {
        'subscriber_change': subscriber_change,
        'view_change': view_change,
        'video_change': video_change,
        'estimated_revenue': estimated_revenue
    }
```

#### 1.2 排行榜计算算法
```python
def calculate_rankings(timeframe='daily'):
    """
    计算排行榜的核心算法
    """
    if timeframe == 'daily':
        # 日排行榜：使用当天数据
        date_filter = date.today()
        revenue_query = db.session.query(
            Channel.name,
            ChannelData.estimated_revenue
        ).join(ChannelData).filter(
            ChannelData.date == date_filter,
            Channel.is_active == True
        ).order_by(ChannelData.estimated_revenue.desc())
        
        subscriber_query = db.session.query(
            Channel.name,
            ChannelData.subscriber_change
        ).join(ChannelData).filter(
            ChannelData.date == date_filter,
            Channel.is_active == True
        ).order_by(ChannelData.subscriber_change.desc())
        
    else:  # monthly
        # 月排行榜：使用30天数据汇总
        date_filter = date.today() - timedelta(days=30)
        revenue_query = db.session.query(
            Channel.name,
            db.func.sum(ChannelData.estimated_revenue).label('total_revenue')
        ).join(ChannelData).filter(
            ChannelData.date >= date_filter,
            Channel.is_active == True
        ).group_by(Channel.id).order_by(db.text('total_revenue DESC'))
        
        subscriber_query = db.session.query(
            Channel.name,
            db.func.sum(ChannelData.subscriber_change).label('total_change')
        ).join(ChannelData).filter(
            ChannelData.date >= date_filter,
            Channel.is_active == True
        ).group_by(Channel.id).order_by(db.text('total_change DESC'))
    
    return {
        'revenue_rankings': revenue_query.limit(10).all(),
        'subscriber_rankings': subscriber_query.limit(10).all()
    }
```
