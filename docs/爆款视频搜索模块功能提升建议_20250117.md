## 🔧 YouTube API 实现限制讨论

### 用户反馈
用户指出建议的高级筛选参数（如engagement_rate、growth_trend、channel_size等）无法直接通过YouTube Data API v3查询。

### API实际支持的参数
```python
# YouTube Search API 实际支持的参数
{
    "q": "搜索关键词",
    "publishedAfter": "2024-01-01T00:00:00Z",  # 时间范围
    "publishedBefore": "2024-12-31T23:59:59Z",
    "order": "relevance/date/rating/viewCount/title",
    "videoDuration": "short/medium/long",  # 视频长度
    "type": "video",
    "maxResults": 50,
    "regionCode": "US",  # 地区筛选 ✅
    "relevanceLanguage": "en"  # 语言筛选 ✅
}
```

### 需要二次处理的参数实现方案

#### 1. 播放量、点赞率、评论率筛选
```python
# 实现思路：两步查询
# 第一步：搜索获取视频ID列表
search_response = youtube.search().list(
    q=keywords,
    type="video",
    maxResults=50
).execute()

# 第二步：获取详细统计数据
video_ids = [item['id']['videoId'] for item in search_response['items']]
stats_response = youtube.videos().list(
    part="statistics,snippet",
    id=','.join(video_ids)
).execute()

# 第三步：筛选和计算
for video in stats_response['items']:
    views = int(video['statistics']['viewCount'])
    likes = int(video['statistics'].get('likeCount', 0))
    comments = int(video['statistics'].get('commentCount', 0))
    
    # 计算点赞率
    engagement_rate = (likes + comments) / views if views > 0 else 0
    
    # 应用筛选条件
    if engagement_rate >= min_engagement_rate:
        # 符合条件的视频
        pass
```

### 优化实现策略

#### 方案1：智能预处理
```python
async def enhanced_search(search_params):
    # 1. 基础搜索
    videos = await basic_youtube_search(search_params)
    
    # 2. 批量获取详细数据
    enhanced_videos = await batch_get_video_details(videos)
    
    # 3. 应用高级筛选
    filtered_videos = apply_advanced_filters(enhanced_videos, search_params)
    
    # 4. 计算评分
    scored_videos = calculate_trending_scores(filtered_videos)
    
    return scored_videos

async def batch_get_video_details(videos):
    # 批量获取视频统计、频道信息等
    video_ids = [v['id'] for v in videos]
    
    # 分批处理，避免API限制
    batch_size = 50
    all_details = []
    
    for i in range(0, len(video_ids), batch_size):
        batch = video_ids[i:i+batch_size]
        details = await get_video_stats_batch(batch)
        all_details.extend(details)
    
    return all_details
```

#### 方案2：缓存优化
```python
# 缓存频道信息，避免重复查询
@cache_result(expire=3600)  # 1小时缓存
async def get_channel_info(channel_id):
    return await youtube.channels().list(
        part="statistics,snippet",
        id=channel_id
    ).execute()

# 缓存视频统计信息
@cache_result(expire=1800)  # 30分钟缓存
async def get_video_stats(video_id):
    return await youtube.videos().list(
        part="statistics,snippet",
        id=video_id
    ).execute()
```

#### 方案3：渐进式加载
```python
# 前端先显示基础搜索结果，再异步加载详细数据
async def progressive_search():
    # 1. 快速返回基础结果
    basic_results = await youtube_search_basic(params)
    yield {"type": "basic", "data": basic_results}
    
    # 2. 异步加载详细统计
    for video in basic_results:
        enhanced_video = await get_video_details(video['id'])
        yield {"type": "enhanced", "data": enhanced_video}
```
